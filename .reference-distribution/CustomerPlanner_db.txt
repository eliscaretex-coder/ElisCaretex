<script>
  var DAYS=['Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'];
  var COLOR={Monday:'cust-day-mon',Tuesday:'cust-day-tue',Wednesday:'cust-day-wed',Thursday:'cust-day-thu',Friday:'cust-day-fri',Saturday:'cust-day-sat'};
  var NAMES=['Ballsbridge Hotel & Spa','Riverside Clinic','O\'Connell','Smithfield Market Café','Grand Canal Medical','Dún Laoghaire Leisure Centre','Áras','Temple Bar Bistro','Phoenix Park Lodge','Blackrock'];
  var dot='<span class="cust-dot" style="background:#0ea5a6"></span>';
  function badge(n){return '<span class="cust-order-badge">'+n+'</span>';}

  var html='<table id="cust-planner"><colgroup>';
  DAYS.forEach(function(){html+='<col class="cg-name"><col class="cg-clothes"><col class="cg-act">';});
  html+='</colgroup><thead><tr class="cust-day-row">';
  DAYS.forEach(function(d){html+='<th colspan="3" class="'+COLOR[d]+'">'+d.toUpperCase()+'</th>';});
  html+='</tr><tr class="cust-sub-row">';
  DAYS.forEach(function(d){html+='<th class="'+COLOR[d]+'">CUSTOMER</th><th class="'+COLOR[d]+'">CL</th><th class="'+COLOR[d]+'"></th>';});
  html+='</tr></thead><tbody>';
  for(var i=0;i<8;i++){html+='<tr>';DAYS.forEach(function(d,di){
    var nm=NAMES[(i+di)%NAMES.length];
    html+='<td class="cust-td-name" style="background:#eef2ff">'+badge(i+1)+dot+nm+'</td>'+
          '<td class="cust-td-num" style="background:#eef2ff">'+((i+di)%3+1)+'</td>'+
          '<td class="cust-td-act" style="background:#eef2ff"><div class="cust-act-wrap">'+
          '<button class="cust-ic-btn cust-ic-move" title="Arrastar">&#8942;</button>'+
          '<button class="cust-ic-btn cust-ic-edit" title="Editar">&#9999;</button>'+
          '<button class="cust-ic-btn cust-ic-del" title="Excluir">&#10005;</button>'+
          '</div></td>';
  });html+='</tr>';}
  html+='</tbody></table>';
  document.getElementById('cust-planner-wrap').innerHTML=html;

  var outer=document.getElementById('cust-scroll-outer'),wrap=document.getElementById('cust-planner-wrap');
  function upd(){var sl=wrap.scrollLeft,max=wrap.scrollWidth-wrap.clientWidth;
    outer.classList.toggle('can-left',sl>4);outer.classList.toggle('can-right',sl<max-4);}
  wrap.addEventListener('scroll',upd);
  var step=Math.max(420,Math.floor(wrap.clientWidth*0.75));
  document.getElementById('cust-scroll-arrow-l').onclick=function(){wrap.scrollLeft-=step;};
  document.getElementById('cust-scroll-arrow-r').onclick=function(){wrap.scrollLeft+=step;};
  wrap.addEventListener('wheel',function(e){if(Math.abs(e.deltaY)>Math.abs(e.deltaX)){wrap.scrollLeft+=e.deltaY;e.preventDefault();}},{passive:false});
  setTimeout(upd,30);
</script>
