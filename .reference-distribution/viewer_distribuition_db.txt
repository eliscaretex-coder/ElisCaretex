var DAYS     = ['mon','tue','wed','thu','fri','sat','sun'];
var DAY_S    = ['Mon','Tue','Wed','Thu','Fri','Sat','Sun'];
var DAY_FULL = ['Monday','Tuesday','Wednesday','Thursday','Friday','Saturday','Sunday'];
var STATUS   = ['Off','Hols','Training','Absent','Sick'];
var APP      = null;

/* ── Helpers ── */
function esc(s){ return String(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }

function tc(hex) {
  try {
    var h=String(hex||'').replace('#','');
    var r=parseInt(h.slice(0,2),16),g=parseInt(h.slice(2,4),16),b=parseInt(h.slice(4,6),16);
    return (0.299*r+0.587*g+0.114*b)/255>0.55?'#000':'#fff';
  } catch(e){ return '#fff'; }
}

function fmtFull(ds){ var p=String(ds||'').split('-'); return p[2]+'/'+p[1]+'/'+p[0]; }
function fmtS(ds)   { var p=String(ds||'').split('-'); return p[2]+'/'+p[1]; }

function getWeekDates(ws) {
  var d = new Date(ws+'T12:00:00Z'), out = [];
  for (var i=0;i<7;i++) { var x=new Date(d); x.setUTCDate(d.getUTCDate()+i); out.push(x.toISOString().slice(0,10)); }
  return out;
}

function todayIdx() { var d=new Date().getUTCDay(); return d===0?6:d-1; }

function parseCell(val) {
  if (!val || STATUS.indexOf(val)!==-1) return {route:val||'',vehicle:''};
  var i=val.indexOf('|');
  return i===-1 ? {route:val,vehicle:''} : {route:val.slice(0,i),vehicle:val.slice(i+1)};
}

function isSecondTurn(assignments, dayKey, driverId, vehicle) {
  var target=String(vehicle||'').replace(/\s+/g,'').toUpperCase();
  if(!target) return false;
  var users=[];
  for(var i=0;i<(assignments||[]).length;i++){
    var p=parseCell((assignments[i]||{})[dayKey]||'');
    if(String(p.vehicle||'').replace(/\s+/g,'').toUpperCase()===target) users.push(assignments[i]);
  }
  for(var j=0;j<users.length;j++) if(String(users[j].driverId)===String(driverId)) return j>0;
  return false;
}

function dayMatch(cd, dayKey) {
  var s=(cd||'').toLowerCase().trim();
  var fl=(DAY_FULL[DAYS.indexOf(dayKey)]||'').toLowerCase();
  var fl3=fl.slice(0,3);
  return s===fl||s===fl3||s.startsWith(fl3)||fl.startsWith(s.slice(0,3));
}

function findRoute(val) {
  var route=parseCell(val).route;
  if (!route||!APP) return null;
  var rs=APP.routes||[];
  for (var i=0;i<rs.length;i++) if (rs[i].displayName===route) return rs[i];
  for (var j=0;j<rs.length;j++) if (rs[j].route===route) return rs[j];
  return null;
}

function getStops(raw, dayKey) {
  var rt=findRoute(raw); if (!rt) return [];
  var out=[], cs=APP.customers||[];
  for (var i=0;i<cs.length;i++) {
    var c=cs[i];
    if (String(c.route).trim()===String(rt.route).trim()&&dayMatch(c.day,dayKey)) out.push(c);
  }
  // Collection stops with no explicit order go to the end,
  // keeping the supervisor's numbered sequence intact for the driver.
  out.sort(function(a,b){
    var ao=(a.order&&a.order>0)?a.order:9999;
    var bo=(b.order&&b.order>0)?b.order:9999;
    return ao-bo;
  });
  // Re-number sequentially so the driver always sees 1..N with no gaps.
  for (var j=0;j<out.length;j++){ out[j]=Object.assign({},out[j],{order:j+1}); }
  return out;
}

function hasStops(raw,dayKey){ return getStops(raw,dayKey).length>0; }

  for (var gi=0;gi<DAYS.length;gi++) {
    var dayKey = DAYS[gi];
    var raw    = a[dayKey]||'';
    var p      = parseCell(raw);
    var rt     = raw&&STATUS.indexOf(p.route)===-1 ? findRoute(raw) : null;
    var bg     = rt ? rt.colour : null;
    var colour = bg ? tc(bg) : null;
    var dn     = rt ? rt.displayName : p.route;
    var label  = dn && dn.length>10 ? dn.slice(0,9)+'.' : dn;
    var stopsOk = rt ? hasStops(raw, dayKey) : false;
    var isT     = gi===tIdx;

    h += '<div class="focus-day'+(isT?' tod':'')+'"><div class="fd-name">'+DAY_S[gi]+'</div><div class="fd-date">'+fmtS(dates[gi])+'</div>';

    var fdCl = 'fd-badge';
    if (!p.route) {
      h += '<div class="'+fdCl+' fd-b-empty noop">&ndash;</div>';
    } else if (p.route==='Off') {
      h += '<div class="'+fdCl+' fd-b-off noop">Off</div>';
    } else if (p.route==='Hols') {
      h += '<div class="'+fdCl+' fd-b-hols noop">Hols</div>';
    } else if (p.route==='Training') {
      h += '<div class="'+fdCl+' fd-b-train noop">Training</div>';
    } else if (p.route==='Absent') {
      h += '<div class="'+fdCl+' fd-b-absent noop">Absent</div>';
    } else if (p.route==='Sick') {
      h += '<div class="'+fdCl+' fd-b-sick noop">Sick</div>';
    } else {
      var clickAttr = 'onclick="openSheet(\''+encodeURIComponent(raw)+'\',\''+dayKey+'\',\''+encodeURIComponent(drv.name)+'\')"';
      var outline   = stopsOk ? ';outline:2px solid rgba(255,255,255,.3)' : '';
      h += '<button class="'+fdCl+'" style="background:'+bg+';color:'+colour+outline+'" '+clickAttr+'>'
          + esc(label);
      if (p.vehicle) h += '<span class="fd-veh">'+esc(p.vehicle)+(isSecondTurn(_focusAsgns,dayKey,drvId,p.vehicle)?' <span class="turn2">2T</span>':'')+'</span>';
      h += '</button>';
    }
    if (stopsOk && p.route&&STATUS.indexOf(p.route)===-1) h += '<div class="fd-sdot"></div>';
    if (isT) h += '<div class="fd-today-ring"></div>';
    h += '</div>'; // focus-day
  }
  h += '</div></div>'; // focus-days, focus-week

  document.getElementById('focus-content').innerHTML = h;
  document.getElementById('focus-view').classList.add('open');
  document.getElementById('focus-view').scrollTop = 0;
}

function openSheet(rawEnc, dayKey, nameEnc) {
  var raw    = decodeURIComponent(rawEnc);
  var drvName= decodeURIComponent(nameEnc);
  var p      = parseCell(raw);
  var dayIdx = DAYS.indexOf(dayKey);
  var dayFull= DAY_FULL[dayIdx]||dayKey;
  var dates  = APP ? getWeekDates(_focusWeekSt||APP.weekStart) : [];
  var dayDisp= DAY_S[dayIdx]+(dates[dayIdx]?' '+fmtS(dates[dayIdx]):'');
  var rt     = findRoute(raw);
  var bg     = rt ? rt.colour : '#3b82f6';
  var colour = tc(bg);
  var stops  = getStops(raw, dayKey);
  var dispName = rt ? rt.displayName : p.route;
  var routeNum = rt ? rt.route : p.route;

  var externalRoute = !!(rt && String(rt.routeType||'STANDARD').toUpperCase()==='EXTERNAL');
  document.getElementById('sht-title').innerHTML = esc(dispName)
    + '<span style="margin-left:8px;background:#1e3a5f;color:#60a5fa;padding:2px 8px;border-radius:5px;font-size:12px;font-weight:700">Route '+esc(routeNum)+'</span>'
    + (externalRoute?'<span style="margin-left:6px;background:#4c1d95;color:#ddd6fe;border:1px solid #8b5cf6;padding:2px 8px;border-radius:5px;font-size:10px;font-weight:900;text-transform:uppercase">External delivery</span>':'');
  document.getElementById('sht-sub').textContent = dayFull+' · '+dayDisp+(drvName?' · '+drvName:'')+(p.vehicle?' · '+p.vehicle:'');

  var body='';
  if (!stops.length) {
    body='<div class="no-stops"><div class="icon">&#128203;</div><h3>No route details</h3><p>No stops for Route '+esc(routeNum)+' on '+dayFull+'.<br>Contact your supervisor.</p></div>';
  } else {
    for (var i=0;i<stops.length;i++) {
      var s=stops[i];
      var hasTrC=s.trolleyClothes&&s.trolleyClothes!=='.';
      var hasTrM=s.trolleyMop&&s.trolleyMop!=='.';
      var hasEir=s.eircode&&s.eircode!=='.';
      var hasKg=s.kg>0;
      var hasNote=s.notes&&s.notes!=='.';
      var overlayKind=s.overlay?String(s.overlay.kind||'').toUpperCase():'';
      var isCollection=(overlayKind==='COLLECTION'||overlayKind==='TEMP_STOP');
      var isExternalDelivery=(overlayKind==='EXTERNAL_DELIVERY');
      body+='<div class="stop'+(isCollection?' collection-stop':'')+(isExternalDelivery?' external-delivery-stop':'')+'"><div class="stop-row"><div class="stop-n">'+s.order+'</div><div class="stop-main">';
      if (isCollection) body+='<div class="collection-badge">&#128666; Collection only</div>';
      if (isExternalDelivery) body+='<div class="external-delivery-badge">&#128230; External delivery</div>';
      if (s.time) body+='<div class="stop-time">&#128336; '+esc(s.time)+'</div>';
      body+='<div class="stop-name">'+esc(s.customer)+'</div></div></div>';
      if (hasEir||hasKg) {
        body+='<div class="stop-chips">';
        if (hasEir) {
          var mapsUrl = 'https://www.google.com/maps/search/?api=1&query=' + encodeURIComponent(s.eircode + ', Ireland');
          body += '<a href="' + mapsUrl + '" target="_blank" rel="noopener" style="text-decoration:none">'
            + '<span class="chip" style="background:#1e3a5f;color:#60a5fa;border:1px solid #2563eb;cursor:pointer;-webkit-tap-highlight-color:transparent">'
            + '<svg viewBox="0 0 24 24"><path d="M21 10c0 7-9 13-9 13s-9-6-9-13a9 9 0 0 1 18 0z"/><circle cx="12" cy="10" r="3"/></svg>'
            + esc(s.eircode) + ' &#x2197;</span></a>';
        }
        if (hasKg)  body+='<span class="chip"><svg viewBox="0 0 24 24"><line x1="12" y1="20" x2="12" y2="10"/><path d="M6 10V4h12v6"/><rect x="2" y="10" width="20" height="10" rx="1"/></svg>'+s.kg.toFixed(1)+' KG</span>';
        body+='</div>';
      }
      if (hasTrC||hasTrM) {
        body+='<div class="trolleys">';
        if (hasTrC) body+='<span class="tr-c">&#129082; Clothes: '+esc(s.trolleyClothes)+'</span>';
        if (hasTrM) body+='<span class="tr-m">&#129529; Mop: '+esc(s.trolleyMop)+'</span>';
        body+='</div>';
      }
      if (hasNote) body+='<div class="stop-note">&#128221; '+esc(s.notes)+'</div>';
      body+='</div>';
    }
  }
  document.getElementById('sht-body').innerHTML = body;
  document.getElementById('sbg').classList.add('open');
  requestAnimationFrame(function(){ document.getElementById('sht').classList.add('open'); });
}

function renderTeamCalendar(){
  var MONTHS=['January','February','March','April','May','June','July','August','September','October','November','December'];
  var y=_teamCalRef.getFullYear(),m=_teamCalRef.getMonth();
  var mEl=document.getElementById('teamcal-month');
  if(mEl) mEl.textContent=MONTHS[m]+' '+y;

  // Map date -> {count, hasHoliday, hasDayoff}
  var map={};
  teamApprovedList().forEach(function(r){
    var d=String(r.LeaveDate||r.StartDate||'').slice(0,10);
    if(!d)return;
    if(!map[d])map[d]={count:0,hol:false,off:false};
    map[d].count++;
    var t=String(r.LeaveType||'').toLowerCase();
    if(t.indexOf('holiday')!==-1)map[d].hol=true;else map[d].off=true;
  });

  var grid=document.getElementById('teamcal-grid');
  if(!grid)return;
  var DOW=['Mon','Tue','Wed','Thu','Fri','Sat','Sun'];
  var html=DOW.map(function(d){return '<div class="mycal-dow">'+d+'</div>';}).join('');

  var startDow=(new Date(y,m,1).getDay()+6)%7;
  var daysInMonth=new Date(y,m+1,0).getDate();
  var todayStr=new Date().toISOString().slice(0,10);

  for(var i=0;i<startDow;i++) html+='<div class="mycal-cell blank"></div>';
  for(var day=1;day<=daysInMonth;day++){
    var ds=y+'-'+String(m+1).padStart(2,'0')+'-'+String(day).padStart(2,'0');
    var info=map[ds];
    var cls='mycal-cell clickable';
    if(info){cls+=' has '+(info.hol?'holiday':'dayoff');}
    if(ds===todayStr)cls+=' today';
    if(ds===_teamSelDay)cls+=' sel';
    var badge=info?('<span style="position:absolute;top:2px;right:4px;font-size:8px;font-weight:900;color:#fff;background:#9d174d;border-radius:999px;padding:0 4px;line-height:1.4">'+info.count+'</span>'):'';
    html+='<div class="'+cls+'" onclick="teamSelect(\''+ds+'\')">'+day+badge+'</div>';
  }
  grid.innerHTML=html;
  renderTeamDay();
}

    rows.map(function(r){
      var t=String(r.LeaveType||'').toLowerCase();
      var cls=t.indexOf('holiday')!==-1?'holiday':'dayoff';
      var label=t.indexOf('holiday')!==-1?'Holiday':(r.LeaveType||'Day off');

    }).join('');
}

function renderMyCalendar(){
  var MONTHS=['January','February','March','April','May','June','July','August','September','October','November','December'];
  var y=_myCalRef.getFullYear(),m=_myCalRef.getMonth();
  document.getElementById('mycal-month').textContent=MONTHS[m]+' '+y;

  var map={};
  (_myCalData.approved||[]).forEach(function(a){
    var t=String(a.leaveType||'').toLowerCase();
    map[a.date]=t.indexOf('holiday')!==-1?'holiday':'dayoff';
  });
  (_myCalData.requests||[]).forEach(function(req){
    if(String(req.status||'').toLowerCase()!=='pending')return;
    eachDate(req.startDate,req.endDate,function(d){ if(!map[d]) map[d]='pending'; });
  });

  var grid=document.getElementById('mycal-grid');
  var DOW=['Mon','Tue','Wed','Thu','Fri','Sat','Sun'];
  var html=DOW.map(function(d){return '<div class="mycal-dow">'+d+'</div>';}).join('');

  var first=new Date(y,m,1);
  var startDow=(first.getDay()+6)%7;
  var daysInMonth=new Date(y,m+1,0).getDate();
  var todayStr=new Date().toISOString().slice(0,10);

  for(var i=0;i<startDow;i++) html+='<div class="mycal-cell blank"></div>';
  for(var day=1;day<=daysInMonth;day++){
    var ds=y+'-'+String(m+1).padStart(2,'0')+'-'+String(day).padStart(2,'0');
    var cls='mycal-cell '+(map[ds]||'')+(ds===todayStr?' today':'');

  }
  grid.innerHTML=html;
}

function renderMyRequests(){

           '<div class="myreq-sub">'+sub+'</div></div>'+

  }).join('');

function eachDate(start,end,fn){
  if(!start)return;
  var d=new Date(start+'T12:00:00Z'),e=new Date((end||start)+'T12:00:00Z');
  var guard=0;
  while(d<=e&&guard<400){fn(d.toISOString().slice(0,10));d.setUTCDate(d.getUTCDate()+1);guard++;}
}

function checkLeaveRequest(){
  clearTimeout(_leaveCheckTimer);
  _leaveCheckTimer=setTimeout(function(){
    var p=leavePayload();
    var btn=document.getElementById('lr-submit');
    btn.disabled=true;
    if(!p.DriverID||!p.StartDate||!p.EndDate){setLeaveMsg('','');return;}
    if(p.EndDate<p.StartDate){setLeaveMsg('err','End date cannot be before start date.');return;}
    setLeaveMsg('warn','Checking availability...');
    google.script.run.withSuccessHandler(function(r){
      if(!r||!r.ok){setLeaveMsg('err',esc((r&&r.msg)||'Could not check these dates.'));return;}
      if(!r.allowed){
        var items=(r.conflicts||[]).slice(0,3).map(function(c){return esc(c.date||'')+' — '+esc(c.message||c.type||'Conflict');}).join('<br>');
        setLeaveMsg('err','These dates have a conflict:<br>'+items);btn.disabled=true;return;
      }
      var warn=(r.warnings&&r.warnings.length)?'<br>Note: there is already a pending request in this range.':'';
      setLeaveMsg('ok','Available for request · '+(r.days||1)+' day'+((r.days||1)!==1?'s':'')+warn);
      btn.disabled=false;
    }).withFailureHandler(function(e){
      setLeaveMsg('err',esc(e.message||'Error checking dates.'));
    }).checkDriverLeaveRequestPublic(p);
  },300);
}

function submitLeaveRequest(){
  var p=leavePayload();
  var btn=document.getElementById('lr-submit');
  var prev=btn.innerHTML;
  btn.disabled=true;
  btn.innerHTML='<span class="lf-spin"></span>Sending…';
  setLeaveMsg('warn','Sending request...');
  google.script.run.withSuccessHandler(function(r){
    btn.innerHTML=prev;
    if(!r||r.ok===false||r.success===false){
      setLeaveMsg('err',esc((r&&r.msg)||(r&&r.message)||'Could not send request.'));
      checkLeaveRequest();
      return;
    }
    var txt='Your supervisor will review it before it is added to the calendar.';
    if(r.emailed===false) txt+=' (Notification email pending — your request is still saved.)';
    document.getElementById('lf-done-text').textContent=txt;
    document.getElementById('lf-fields').style.display='none';
    document.getElementById('lf-done').classList.add('show');
  }).withFailureHandler(function(e){
    btn.innerHTML=prev;
    setLeaveMsg('err',esc(e.message||'Error sending request.'));
    checkLeaveRequest();
  }).submitDriverLeaveRequestPublic(p);
}
